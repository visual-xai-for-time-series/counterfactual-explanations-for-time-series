# Dependencies for cf_codec (CoDec: Counterfactual Decomposition).
#
# The core algorithm (decompositions.py / matching.py / references.py /
# perturbation.py / codec.py) only needs the first two blocks below -
# everything else already ships in the repository root's requirements.txt,
# this file just scopes it down so CoDec can be installed/run on its own:
#
#   pip install -r cfts/cf_codec/requirements.txt
#
# The third block is only needed for cfts/cf_codec/experiments/compare_ucr.py
# (dataset loading, classifier training, results table/plot).

# --- Core (decompositions.py, matching.py, references.py, perturbation.py, codec.py) ---
numpy >= 2.0.1
scipy >= 1.14.1
torch >= 2.3.1

# --- Decomposition-specific (decompositions.py) ---
# "fourier" and "quantile" decomposition need no extra dependency beyond
# scipy/numpy above; "shapelet" and "eigen" (SSA) are pure numpy too.
emd-signal >= 0.6.0    # decomposition="emd" (the IMFACT-compatible baseline)
PyWavelets >= 1.8.0     # decomposition="wavelet"
statsmodels >= 0.14.6   # decomposition="stl" (periodic branch; falls back to "fourier" without it)
ruptures >= 1.1.10      # decomposition="changepoint"

# --- experiments/compare_ucr.py only ---
aeon >= 1.3.0            # UCR/UEA dataset loading (load_classification)
scikit-learn >= 1.5.1     # OneHotEncoder, f1_score
pandas >= 2.3.0           # results/summary tables
matplotlib >= 3.10.8      # bar_metrics.png
